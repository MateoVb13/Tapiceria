
DROP SCHEMA IF EXISTS `tapiceria` ;
CREATE SCHEMA IF NOT EXISTS `tapiceria` DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci ;
USE `tapiceria` ;

-- -----------------------------------------------------
-- Table `tapiceria`.`usuarios`
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `tapiceria`.`usuarios` (
`IdUsuario` INT NOT NULL AUTO_INCREMENT,
`NombreUsuario` VARCHAR(100) NOT NULL,
`Correo` VARCHAR(100) NOT NULL,
`Contrasena` VARCHAR(255) NOT NULL, -- Almacena el HASH seguro de la contraseña
PRIMARY KEY (`IdUsuario`),
UNIQUE INDEX `Correo_UNIQUE` (`Correo` ASC) VISIBLE) -- Añadir índice UNIQUE al correo
ENGINE = InnoDB
DEFAULT CHARACTER SET = utf8mb4
COLLATE = utf8mb4_0900_ai_ci; -- Usar la collación del esquema


-- -----------------------------------------------------
-- Table `tapiceria`.`clientes`
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `tapiceria`.`clientes` (
  `IdCliente` INT NOT NULL AUTO_INCREMENT,
  `IdUsuario` INT NULL DEFAULT NULL, -- Puede ser NULL si un cliente no tiene cuenta de usuario
  `NombreCompleto` VARCHAR(100) NOT NULL,
  `Contacto` VARCHAR(100) NULL DEFAULT NULL, 
  `Direccion` VARCHAR(200) NULL DEFAULT NULL,
  PRIMARY KEY (`IdCliente`),
  INDEX `fk_clientes_usuarios_idx` (`IdUsuario` ASC) VISIBLE,
  CONSTRAINT `fk_clientes_usuarios`
    FOREIGN KEY (`IdUsuario`)
    REFERENCES `tapiceria`.`usuarios` (`IdUsuario`)
    ON DELETE SET NULL -- Si se elimina el usuario, la FK en cliente se pone a NULL
    ON UPDATE NO ACTION
)
ENGINE = InnoDB
DEFAULT CHARACTER SET = utf8mb4
COLLATE = utf8mb4_0900_ai_ci;

-- -----------------------------------------------------
-- Table `tapiceria`.`servicios`
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `tapiceria`.`servicios` (
`IdServicio` INT NOT NULL AUTO_INCREMENT,
`Descripcion` VARCHAR(150) NOT NULL,
`DuracionEstimada` INT NOT NULL COMMENT 'Duración en minutos', -- Comentario útil
`Precio` DECIMAL(10,2) NOT NULL,
`Categoria` VARCHAR(50) NULL DEFAULT NULL,
PRIMARY KEY (`IdServicio`))
ENGINE = InnoDB
DEFAULT CHARACTER SET = utf8mb4
COLLATE = utf8mb4_0900_ai_ci; -- Usar la collación del esquema


-- -----------------------------------------------------
-- Table `tapiceria`.`empleados`
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `tapiceria`.`empleados` (
`IdEmpleado` INT NOT NULL AUTO_INCREMENT,
`NombreCompleto` VARCHAR(100) NOT NULL,
`Especialidad` VARCHAR(100) NULL DEFAULT NULL,
`Contacto` VARCHAR(100) NULL DEFAULT NULL,
PRIMARY KEY (`IdEmpleado`))
ENGINE = InnoDB
DEFAULT CHARACTER SET = utf8mb4
COLLATE = utf8mb4_0900_ai_ci; -- Usar la collación del esquema


-- -----------------------------------------------------
-- Table `tapiceria`.`empleado_disponibilidad`
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `tapiceria`.`empleado_disponibilidad` (
  `IdEmpleadoDisponibilidad` INT NOT NULL AUTO_INCREMENT,
  `IdEmpleado` INT NOT NULL, -- Clave foránea a la tabla empleados
  `DiaSemana` INT NOT NULL COMMENT '1=Lunes, 7=Domingo', -- O podrías usar un ENUM o VARCHAR
  `HoraInicio` TIME NOT NULL, -- Solo la hora (ej: '09:00:00')
  `HoraFin` TIME NOT NULL, -- Solo la hora (ej: '17:00:00')
  PRIMARY KEY (`IdEmpleadoDisponibilidad`),
  INDEX `fk_empleado_disponibilidad_empleados1_idx` (`IdEmpleado` ASC) VISIBLE,
  CONSTRAINT `fk_empleado_disponibilidad_empleados1`
    FOREIGN KEY (`IdEmpleado`)
    REFERENCES `tapiceria`.`empleados` (`IdEmpleado`)
    ON DELETE CASCADE -- Si eliminas al empleado, elimina sus registros de disponibilidad
    ON UPDATE NO ACTION)
ENGINE = InnoDB
DEFAULT CHARACTER SET = utf8mb4
COLLATE = utf8mb4_0900_ai_ci; -- Usar la collación del esquema


-- -----------------------------------------------------
-- Table `tapiceria`.`citas`
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `tapiceria`.`citas` (
`IdCita` INT NOT NULL AUTO_INCREMENT,
`IdCliente` INT NOT NULL,
`IdServicio` INT NOT NULL,
`IdEmpleado` INT NULL DEFAULT NULL, -- Todavía permite asignar UN empleado. Para múltiples, necesitas tabla intermedia cita_empleado.
`FechaInicio` DATETIME NOT NULL,
`FechaFin` DATETIME NOT NULL,
`Estado` VARCHAR(50) NOT NULL COMMENT 'Ej: Pendiente, Confirmada, Completada, Cancelada', -- Comentario útil
`Notas` VARCHAR(300) NULL DEFAULT NULL,
PRIMARY KEY (`IdCita`),
INDEX `fk_citas_clientes1_idx` (`IdCliente` ASC) VISIBLE,
INDEX `fk_citas_servicios1_idx` (`IdServicio` ASC) VISIBLE,
INDEX `fk_citas_empleados1_idx` (`IdEmpleado` ASC) VISIBLE,
CONSTRAINT `fk_citas_clientes1`
FOREIGN KEY (`IdCliente`)
REFERENCES `tapiceria`.`clientes` (`IdCliente`)
ON DELETE NO ACTION -- O SET NULL/CASCADE si quieres que pase algo con las citas del cliente eliminado
ON UPDATE NO ACTION,
CONSTRAINT `fk_citas_servicios1`
FOREIGN KEY (`IdServicio`)
REFERENCES `tapiceria`.`servicios` (`IdServicio`)
ON DELETE NO ACTION -- O SET NULL/CASCADE si quieres que pase algo con las citas del servicio eliminado
ON UPDATE NO ACTION,
CONSTRAINT `fk_citas_empleados1`
FOREIGN KEY (`IdEmpleado`)
REFERENCES `tapiceria`.`empleados` (`IdEmpleado`)
ON DELETE SET NULL
ON UPDATE NO ACTION)
ENGINE = InnoDB
DEFAULT CHARACTER SET = utf8mb4
COLLATE = utf8mb4_0900_ai_ci; -- Usar la collación del esquema


-- -----------------------------------------------------
-- Table `tapiceria`.`pagos`
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `tapiceria`.`pagos` (
`IdPago` INT NOT NULL AUTO_INCREMENT,
`IdCita` INT NOT NULL,
`TipoPago` VARCHAR(50) NOT NULL COMMENT 'Ej: Efectivo, Tarjeta, Transferencia', -- Comentario útil
`ValorPago` DECIMAL(10,2) NULL DEFAULT NULL, -- Añadimos ValorPago como sugerido en el diagrama
PRIMARY KEY (`IdPago`),
INDEX `fk_pagos_citas1_idx` (`IdCita` ASC) VISIBLE,
CONSTRAINT `fk_pagos_citas1`
FOREIGN KEY (`IdCita`)
REFERENCES `tapiceria`.`citas` (`IdCita`)
ON DELETE CASCADE -- Si se elimina la cita, elimina los pagos asociados
 ON UPDATE NO ACTION)
ENGINE = InnoDB
DEFAULT CHARACTER SET = utf8mb4
COLLATE = utf8mb4_0900_ai_ci; -- Usar la collación del esquema

DROP TABLE IF EXISTS `horarios_base_agendamiento`;

-- Creamos la tabla mejorada
CREATE TABLE IF NOT EXISTS `horarios_base_agendamiento` (
  `IdHorarioBase` INT NOT NULL AUTO_INCREMENT,
  `DiaSemana` INT NOT NULL COMMENT 'Día de la semana (1=Lunes, 7=Domingo)',
  `HoraInicio` TIME NOT NULL COMMENT 'Hora de inicio del intervalo',
  `DuracionMinutos` INT NOT NULL COMMENT 'Duración estándar de este intervalo en minutos',
  `IdServicio` INT NULL COMMENT 'Servicio específico al que aplica este horario, NULL para todos',
  `TipoDia` VARCHAR(50) NULL COMMENT 'Ej: "Lunes-Viernes", "Sábado", "Domingo" - Opcional, si prefieres texto',
  `Activo` BOOLEAN NOT NULL DEFAULT TRUE COMMENT 'Indica si este horario base está activo para agendar',
  PRIMARY KEY (`IdHorarioBase`),
  UNIQUE INDEX `UK_DiaHoraServicio` (`DiaSemana` ASC, `HoraInicio` ASC, `IdServicio` ASC) VISIBLE, -- Asegura que no haya horarios base duplicados
  INDEX `fk_horarios_base_servicios1_idx` (`IdServicio` ASC) VISIBLE,
  CONSTRAINT `fk_horarios_base_servicios1`
    FOREIGN KEY (`IdServicio`)
    REFERENCES `tapiceria`.`servicios` (`IdServicio`)
    ON DELETE SET NULL -- Si se elimina el servicio, el horario sigue aplicándose a todos los servicios
    ON UPDATE NO ACTION
) ENGINE = InnoDB;

    IN p_Correo VARCHAR(100)

    FROM usuarios

Delimiter $$
-- 3. Insertar un nuevo usuario
DROP PROCEDURE IF EXISTS InsertarUsuario$$
CREATE PROCEDURE InsertarUsuario(
    IN p_NombreUsuario VARCHAR(100),
    IN p_Correo VARCHAR(100),
    IN p_Contrasena VARCHAR(255) -- Espera el HASH seguro
)
BEGIN
    -- Considerar añadir manejo de errores si Correo o NombreUsuario ya existen (UNIQUE constraint ayuda)
    INSERT INTO usuarios (NombreUsuario, Correo, Contrasena)
    VALUES (p_NombreUsuario, p_Correo, p_Contrasena);

    -- Devolver el Id generado
    SELECT LAST_INSERT_ID() AS NuevoIdUsuario;
END$$
Delimiter $$
-- 4. Actualizar un usuario existente
DROP PROCEDURE IF EXISTS ActualizarUsuario$$
CREATE PROCEDURE ActualizarUsuario(
    IN p_IdUsuario INT,
    IN p_NombreUsuario VARCHAR(100),
    IN p_Correo VARCHAR(100),
    IN p_Contrasena VARCHAR(255) -- Espera el HASH seguro si se actualiza
)
BEGIN
    UPDATE usuarios
    SET
        NombreUsuario = p_NombreUsuario,
        Correo = p_Correo,
        Contrasena = p_Contrasena
    WHERE IdUsuario = p_IdUsuario;

END$$
Delimiter $$
-- 5. Eliminar un usuario por su Id
DROP PROCEDURE IF EXISTS EliminarUsuario$$
CREATE PROCEDURE EliminarUsuario(
    IN p_IdUsuario INT
)
BEGIN
    -- ON DELETE SET NULL/CASCADE en las FKs (clientes) maneja las dependencias
    DELETE FROM usuarios
    WHERE IdUsuario = p_IdUsuario;

END$$

-- 6. Procedimiento de Login (Validando Correo y Contraseña HASH)
-- Valida que el correo y la contraseña (hash) coincidan
-- NOTA DE SEGURIDAD: La API debe hashear la contraseña plana antes de llamar a este SP.
DROP PROCEDURE IF EXISTS ValidarLogin$$
CREATE PROCEDURE ValidarLogin(
    IN p_Correo VARCHAR(100),
    IN p_Contrasena VARCHAR(255) -- Espera el HASHED password from the API
)
BEGIN
    SELECT
        IdUsuario,
        NombreUsuario,
        Correo,
        Contrasena
    FROM usuarios
    WHERE Correo = p_Correo
      AND Contrasena = p_Contrasena; -- Compara el hash de entrada con el hash almacenado
END$$

-- 26. Obtener todas las citas (con detalles de cliente, servicio y empleado)

BEGIN
    SELECT
        c.IdCita,
        c.IdCliente,
        cl.NombreCompleto AS NombreCliente,
        c.IdServicio,
        s.Descripcion AS NombreServicio,
        s.DuracionEstimada,
        s.Precio,
        c.IdEmpleado,
        e.NombreCompleto AS NombreEmpleado, -- LEFT JOIN porque IdEmpleado puede ser NULL

    INNER JOIN clientes cl ON c.IdCliente = cl.IdCliente
    INNER JOIN servicios s ON c.IdServicio = s.IdServicio
    LEFT JOIN empleados e ON c.IdEmpleado = e.IdEmpleado;
END$$

-- Verificar la disponibilidad real para un horario base
DROP PROCEDURE IF EXISTS VerificarDisponibilidadHorarioBase$$
CREATE PROCEDURE VerificarDisponibilidadHorarioBase(
    IN p_IdHorarioBase INT,
    IN p_Fecha DATE -- Fecha específica a verificar
)
BEGIN
    DECLARE v_DiaSemana INT;
    DECLARE v_HoraInicio TIME;
    DECLARE v_DuracionMinutos INT;
    DECLARE v_FechaInicio DATETIME;
    DECLARE v_FechaFin DATETIME;
    
    -- Obtener información del horario base
    SELECT 
        DiaSemana, 
        HoraInicio,
        DuracionMinutos
    INTO 
        v_DiaSemana, 
        v_HoraInicio,
        v_DuracionMinutos
    FROM horarios_base_agendamiento
    WHERE IdHorarioBase = p_IdHorarioBase;
    
    -- Calcular fecha y hora completas
    SET v_FechaInicio = TIMESTAMP(p_Fecha, v_HoraInicio);
    SET v_FechaFin = v_FechaInicio + INTERVAL v_DuracionMinutos MINUTE;
    
    -- Devolver empleados disponibles para este horario
    SELECT
        e.IdEmpleado,
        e.NombreCompleto,
        v_FechaInicio AS FechaInicio,
        v_FechaFin AS FechaFin
    FROM empleados e
    -- Unir con la disponibilidad del empleado para ese día de la semana
    INNER JOIN empleado_disponibilidad ed ON e.IdEmpleado = ed.IdEmpleado
    WHERE ed.DiaSemana = v_DiaSemana
      AND ed.HoraInicio <= v_HoraInicio
      AND ed.HoraFin >= ADDTIME(v_HoraInicio, SEC_TO_TIME(v_DuracionMinutos * 60))
      -- Y no tiene citas en ese horario
      AND NOT EXISTS (
          SELECT 1
          FROM citas c
          WHERE c.IdEmpleado = e.IdEmpleado
            AND c.FechaInicio < v_FechaFin
            AND c.FechaFin > v_FechaInicio
      );
END$$

-- Obtener disponibilidad completa (combinando horarios base y empleados)
DROP PROCEDURE IF EXISTS ObtenerDisponibilidadCompleta$$
CREATE PROCEDURE ObtenerDisponibilidadCompleta(
    IN p_IdServicio INT,
    IN p_Fecha DATE
)
BEGIN
    DECLARE v_DiaSemana INT;
    
    -- Calcular el día de la semana
    SET v_DiaSemana = WEEKDAY(p_Fecha) + 1; -- Convertir de 0-6 a 1-7
    
    -- Obtener todos los horarios base disponibles para este servicio en este día
    CREATE TEMPORARY TABLE temp_horarios_disponibles AS
    SELECT
        hb.IdHorarioBase,
        hb.HoraInicio,
        hb.DuracionMinutos,
        TIMESTAMP(p_Fecha, hb.HoraInicio) AS FechaHoraInicio,
        TIMESTAMP(p_Fecha, hb.HoraInicio) + INTERVAL hb.DuracionMinutos MINUTE AS FechaHoraFin
    FROM horarios_base_agendamiento hb
    WHERE (hb.IdServicio = p_IdServicio OR hb.IdServicio IS NULL)
      AND hb.DiaSemana = v_DiaSemana
      AND hb.Activo = TRUE;
    
    -- Para cada horario, encontrar empleados disponibles
    SELECT
        h.FechaHoraInicio AS FechaInicio,
        h.FechaHoraFin AS FechaFin,
        e.IdEmpleado,
        e.NombreCompleto AS NombreEmpleado
    FROM temp_horarios_disponibles h
    CROSS JOIN empleados e
    -- Verificar disponibilidad del empleado en ese día y hora
    INNER JOIN empleado_disponibilidad ed ON e.IdEmpleado = ed.IdEmpleado
    WHERE ed.DiaSemana = v_DiaSemana
      AND ed.HoraInicio <= h.HoraInicio
      AND ed.HoraFin >= ADDTIME(h.HoraInicio, SEC_TO_TIME(h.DuracionMinutos * 60))
      -- Y no tiene citas en ese horario
      AND NOT EXISTS (
          SELECT 1
          FROM citas c
          WHERE c.IdEmpleado = e.IdEmpleado
            AND c.FechaInicio < h.FechaHoraFin
            AND c.FechaFin > h.FechaHoraInicio
      )
    ORDER BY h.FechaHoraInicio, e.NombreCompleto;
    
    -- Eliminar tabla temporal
    DROP TEMPORARY TABLE IF EXISTS temp_horarios_disponibles;
END$$

-- Funcionalidad para crear múltiples horarios base para un rango de tiempo
DROP PROCEDURE IF EXISTS CrearHorariosBaseRango$$
CREATE PROCEDURE CrearHorariosBaseRango(
    IN p_DiaSemana INT, -- 1-7 para el día específico
    IN p_HoraInicio TIME, -- Hora de inicio del rango
    IN p_HoraFin TIME, -- Hora de fin del rango
    IN p_DuracionSlotMinutos INT, -- Duración de cada slot
    IN p_IdServicio INT, -- NULL para todos los servicios
    IN p_TipoDia VARCHAR(50), -- Descripción opcional
    IN p_Activo BOOLEAN -- TRUE/FALSE
)
BEGIN
    DECLARE v_HoraActual TIME;
    DECLARE v_MensajeResultado VARCHAR(100);
    DECLARE v_ContadorCreados INT DEFAULT 0;
    
    SET v_HoraActual = p_HoraInicio;
    
    -- Crear slots de tiempo mientras estemos dentro del rango
    WHILE v_HoraActual < p_HoraFin DO
        -- Intentar insertar el horario base
        INSERT IGNORE INTO horarios_base_agendamiento (
            DiaSemana, 
            HoraInicio, 
            DuracionMinutos, 
            IdServicio, 
            TipoDia, 
            Activo
        )
        VALUES (
            p_DiaSemana,
            v_HoraActual,
            p_DuracionSlotMinutos,
            p_IdServicio,
            p_TipoDia,
            p_Activo
        );
        
        -- Actualizar contador
        SET v_ContadorCreados = v_ContadorCreados + ROW_COUNT();
        
        -- Avanzar al siguiente slot
        SET v_HoraActual = ADDTIME(v_HoraActual, SEC_TO_TIME(p_DuracionSlotMinutos * 60));
    END WHILE;
    
    -- Devolver resultado
    SELECT 
        CONCAT('Se crearon ', v_ContadorCreados, ' horarios base para el día ', p_DiaSemana) AS Resultado;
END$$
